SELECT order_id, SUM(price * quantity) AS total_price FROM products JOIN line_items ON products.product_id = line_items.product_id GROUP BY order_id ORDER BY order_id LIMIT 5;
SELECT customers.customer_name, AVG(order_totals.total_price) AS average_order_price FROM customers JOIN orders ON customers.customer_id = orders.customer_id JOIN (SELECT order_id, SUM(price * quantity) AS total_price FROM line_items JOIN products ON line_items.product_id = products.product_id GROUP BY order_id) AS order_totals ON orders.order_id = order_totals.order_id GROUP BY customers.customer_id, customers.customer_name ORDER BY customers.customer_name;
