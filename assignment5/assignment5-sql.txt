SELECT orders_limited.order_id, SUM(products.price * line_items.quantity) AS total_price FROM (SELECT orders.order_id FROM orders ORDER BY orders.order_id ASC LIMIT 5) AS orders_limited JOIN line_items ON orders_limited.order_id = line_items.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY orders_limited.order_id ORDER BY orders_limited.order_id ASC;
SELECT customers.customer_name, AVG(order_totals.total_price) AS average_order_price FROM customers JOIN (SELECT orders.customer_id, orders.order_id, SUM(products.price * line_items.quantity) AS total_price FROM orders JOIN line_items ON orders.order_id = line_items.order_id JOIN products ON line_items.product_id = products.product_id GROUP BY orders.customer_id, orders.order_id) AS order_totals ON customers.customer_id = order_totals.customer_id GROUP BY customers.customer_id, customers.customer_name ORDER BY customers.customer_name ASC;
